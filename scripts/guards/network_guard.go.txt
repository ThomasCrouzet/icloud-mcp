package main

import (
  "fmt"
  "go/ast"
  "go/parser"
  "go/token"
  "io/fs"
  "net"
  "net/url"
  "os"
  "path/filepath"
  "regexp"
  "strconv"
  "strings"
)

var (
  calendarShard = regexp.MustCompile(`^p[0-9]{1,3}-caldav[.]icloud[.]com$`)
  contactsShard = regexp.MustCompile(`^p[0-9]{1,3}-contacts[.]icloud[.]com$`)
  hostLiteral = regexp.MustCompile(`^(?:[A-Za-z0-9-]+[.])+[A-Za-z]{2,}(?::[0-9]+)?$`)
  urlLiteral = regexp.MustCompile(`https?://[^[:space:]"'<>]+`)
)

func davHostAllowed(host string) bool {
  return host == "caldav.icloud.com" || calendarShard.MatchString(host) ||
    host == "contacts.icloud.com" || contactsShard.MatchString(host)
}

func standaloneHostAllowed(value string) bool {
  host, port := value, ""
  if strings.Contains(value, ":") {
    var err error
    host, port, err = net.SplitHostPort(value)
    if err != nil {
      return false
    }
  }
  if davHostAllowed(host) {
    return port == "" || port == "443"
  }
  switch host {
  case "imap.mail.me.com":
    return port == "" || port == "993"
  case "smtp.mail.me.com":
    return port == "" || port == "587"
  default:
    return false
  }
}

func networkURLAllowed(value string) bool {
  if value == "http://apple.com/ns/ical/" {
    return true
  }
  parsed, err := url.Parse(value)
  return err == nil && parsed.Scheme == "https" && davHostAllowed(parsed.Hostname()) &&
    (parsed.Port() == "" || parsed.Port() == "443")
}

func networkName(name string) bool {
  name = strings.ToLower(name)
  for _, marker := range []string{"url", "host", "addr", "address", "endpoint", "servername"} {
    if strings.Contains(name, marker) {
      return true
    }
  }
  return false
}

func staticString(expression ast.Expr, constants map[string]string) (string, bool) {
  switch value := expression.(type) {
  case *ast.BasicLit:
    if value.Kind != token.STRING {
      return "", false
    }
    decoded, err := strconv.Unquote(value.Value)
    return decoded, err == nil
  case *ast.BinaryExpr:
    if value.Op != token.ADD {
      return "", false
    }
    left, leftOK := staticString(value.X, constants)
    right, rightOK := staticString(value.Y, constants)
    return left + right, leftOK && rightOK
  case *ast.ParenExpr:
    return staticString(value.X, constants)
  case *ast.Ident:
    decoded, ok := constants[value.Name]
    return decoded, ok
  default:
    return "", false
  }
}

func fileConstants(source *ast.File) map[string]string {
  constants := make(map[string]string)
  for pass := 0; pass < len(source.Decls); pass++ {
    changed := false
    for _, declaration := range source.Decls {
      group, ok := declaration.(*ast.GenDecl)
      if !ok || group.Tok != token.CONST {
        continue
      }
      for _, item := range group.Specs {
        spec, ok := item.(*ast.ValueSpec)
        if !ok || len(spec.Names) != len(spec.Values) {
          continue
        }
        for index, name := range spec.Names {
          if _, exists := constants[name.Name]; exists {
            continue
          }
          if value, ok := staticString(spec.Values[index], constants); ok {
            constants[name.Name] = value
            changed = true
          }
        }
      }
    }
    if !changed {
      break
    }
  }
  return constants
}

func importedPackages(source *ast.File) map[string]string {
  packages := make(map[string]string)
  for _, spec := range source.Imports {
    imported, err := strconv.Unquote(spec.Path.Value)
    if err != nil {
      continue
    }
    name := filepath.Base(imported)
    if spec.Name != nil {
      name = spec.Name.Name
    }
    packages[name] = imported
  }
  return packages
}

func directNetworkCall(call *ast.CallExpr, constants, packages map[string]string) (string, bool) {
  selector, ok := call.Fun.(*ast.SelectorExpr)
  if !ok || len(call.Args) == 0 {
    return "", false
  }
  if selector.Sel.Name == "Dial" || selector.Sel.Name == "DialContext" {
    address, literal := staticString(call.Args[len(call.Args)-1], constants)
    if !literal {
      return "unresolved dynamic direct dial target", true
    }
    if literal && !standaloneHostAllowed(address) {
      return fmt.Sprintf("unauthorized direct dial target %q", address), true
    }
    return "", false
  }
  packageName, packageCall := selector.X.(*ast.Ident)
  if !packageCall || packages[packageName.Name] != "net/http" {
    return "", false
  }
  switch selector.Sel.Name {
  case "Get", "Head", "Post", "PostForm":
    target, literal := staticString(call.Args[0], constants)
    if !literal {
      return "unresolved dynamic direct HTTP target", true
    }
    if literal && !networkURLAllowed(target) {
      return fmt.Sprintf("unauthorized direct HTTP target %q", target), true
    }
  }
  return "", false
}

func main() {
  root := "."
  if len(os.Args) == 2 {
    root = os.Args[1]
  }
  files := token.NewFileSet()
  violations := 0
  err := filepath.WalkDir(root, func(path string, entry fs.DirEntry, walkErr error) error {
    if walkErr != nil {
      return walkErr
    }
    if entry.IsDir() {
      if entry.Name() == ".git" || entry.Name() == "vendor" {
        return filepath.SkipDir
      }
      return nil
    }
    if !strings.HasSuffix(path, ".go") {
      return nil
    }
    source, parseErr := parser.ParseFile(files, path, nil, 0)
    if parseErr != nil {
      return parseErr
    }
    for _, spec := range source.Imports {
      imported, unquoteErr := strconv.Unquote(spec.Path.Value)
      if unquoteErr == nil && imported == "os/exec" {
        fmt.Printf("%s: os/exec is forbidden\n", files.Position(spec.Pos()))
        violations++
      }
    }
    if strings.HasSuffix(path, "_test.go") {
      return nil
    }
    constants := fileConstants(source)
    packages := importedPackages(source)
    networkLiterals := make(map[token.Pos]bool)
    ast.Inspect(source, func(node ast.Node) bool {
      spec, ok := node.(*ast.ValueSpec)
      if !ok {
        return true
      }
      guarded := false
      for _, name := range spec.Names {
        guarded = guarded || networkName(name.Name)
      }
      if !guarded {
        return true
      }
      for _, value := range spec.Values {
        ast.Inspect(value, func(child ast.Node) bool {
          if literal, ok := child.(*ast.BasicLit); ok && literal.Kind == token.STRING {
            networkLiterals[literal.Pos()] = true
          }
          return true
        })
      }
      return true
    })
    ast.Inspect(source, func(node ast.Node) bool {
      if call, ok := node.(*ast.CallExpr); ok {
        if message, violation := directNetworkCall(call, constants, packages); violation {
          fmt.Printf("%s: %s\n", files.Position(call.Pos()), message)
          violations++
        }
      }
      if selector, ok := node.(*ast.SelectorExpr); ok && selector.Sel.Name == "InsecureSkipVerify" {
        fmt.Printf("%s: InsecureSkipVerify is forbidden outside tests\n", files.Position(selector.Pos()))
        violations++
      }
      literal, ok := node.(*ast.BasicLit)
      if !ok || literal.Kind != token.STRING {
        return true
      }
      value, unquoteErr := strconv.Unquote(literal.Value)
      if unquoteErr != nil {
        return true
      }
      matches := urlLiteral.FindAllString(value, -1)
      if len(matches) == 0 && (strings.Contains(value, "https://") || strings.Contains(value, "http://")) {
        canonicalBuilder := filepath.ToSlash(path) == "internal/contacts/client.go" && value == "https://"
        if !canonicalBuilder {
          fmt.Printf("%s: incomplete network URL literal is not allowed\n", files.Position(literal.Pos()))
          violations++
        }
      }
      for _, match := range matches {
        if !networkURLAllowed(match) {
          fmt.Printf("%s: unauthorized network URL literal %q\n", files.Position(literal.Pos()), match)
          violations++
        }
      }
      if networkLiterals[literal.Pos()] && hostLiteral.MatchString(value) && !standaloneHostAllowed(value) {
        fmt.Printf("%s: unauthorized network host literal %q\n", files.Position(literal.Pos()), value)
        violations++
      }
      return true
    })
    return nil
  })
  if err != nil {
    fmt.Fprintln(os.Stderr, err)
    os.Exit(1)
  }
  if violations != 0 {
    os.Exit(1)
  }
}
